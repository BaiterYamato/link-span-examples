cmake_minimum_required(VERSION 3.26)
project(LinkSpanShovelDemo LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OOT_NATIVE_SDK "" CACHE FILEPATH "OotNativeSdk.cmake gerado pelo host")
if(NOT EXISTS "${OOT_NATIVE_SDK}")
    message(FATAL_ERROR "Informe -DOOT_NATIVE_SDK=<build-do-host>/native-sdk/Release/OotNativeSdk.cmake")
endif()
include("${OOT_NATIVE_SDK}")

# Header público do NEI (linkspan/nei/nei_items.h): a pasta include/ do pacote do NEI ou do host.
set(LINKSPAN_NEI_INCLUDE "" CACHE PATH "Pasta include/ do NEI (pacote .shipmod extraído ou soh/native-sdk/nei-core/include)")
if(NOT EXISTS "${LINKSPAN_NEI_INCLUDE}/linkspan/nei/nei_items.h")
    message(FATAL_ERROR "Informe -DLINKSPAN_NEI_INCLUDE=<pasta com linkspan/nei/nei_items.h>")
endif()

# package_assets: acha a pasta assets/ do pacote, ao lado de provider/ (ícone e modelo da pá).
set(common_dir "${CMAKE_CURRENT_SOURCE_DIR}/../common")
add_library(linkspan_shovel_demo SHARED shovel_demo.cpp "${common_dir}/package_assets.cpp")
target_include_directories(linkspan_shovel_demo PRIVATE ${OOT_NATIVE_INCLUDE_DIRS} "${LINKSPAN_NEI_INCLUDE}" "${common_dir}")
target_compile_definitions(linkspan_shovel_demo PRIVATE ${OOT_NATIVE_COMPILE_DEFINITIONS})
if(MSVC)
    target_compile_options(linkspan_shovel_demo PRIVATE /utf-8 /Zc:preprocessor)
endif()
set(provider_dir "${CMAKE_BINARY_DIR}/mod/provider")
set_target_properties(linkspan_shovel_demo PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${provider_dir}")
foreach(config Debug Release RelWithDebInfo MinSizeRel)
    string(TOUPPER "${config}" upper)
    set_target_properties(linkspan_shovel_demo PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${upper} "${provider_dir}")
endforeach()
