cmake_minimum_required(VERSION 3.26)
project(LinkSpanFairyLightColor LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OOT_NATIVE_SDK "" CACHE FILEPATH "OotNativeSdk.cmake gerado pelo host")
if(NOT EXISTS "${OOT_NATIVE_SDK}")
    message(FATAL_ERROR "Informe -DOOT_NATIVE_SDK=<build-do-host>/native-sdk/Release/OotNativeSdk.cmake")
endif()
include("${OOT_NATIVE_SDK}")

add_library(linkspan_fairy_light_color SHARED fairy_light_color.cpp)
target_include_directories(linkspan_fairy_light_color PRIVATE ${OOT_NATIVE_INCLUDE_DIRS})
target_compile_definitions(linkspan_fairy_light_color PRIVATE ${OOT_NATIVE_COMPILE_DEFINITIONS})
if(MSVC)
    target_compile_options(linkspan_fairy_light_color PRIVATE /utf-8 /Zc:preprocessor /W4 /WX)
endif()
set(provider_dir "${CMAKE_BINARY_DIR}/mod/provider")
set_target_properties(linkspan_fairy_light_color PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${provider_dir}")
foreach(config Debug Release RelWithDebInfo MinSizeRel)
    string(TOUPPER "${config}" upper)
    set_target_properties(linkspan_fairy_light_color PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${upper} "${provider_dir}")
endforeach()
