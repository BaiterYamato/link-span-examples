cmake_minimum_required(VERSION 3.26)
project(FogreMaster LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(FOG_FIELD_TESTS_ONLY "Compilar somente o campo CPU, sem SDK/provider" OFF)
option(FOG_STABLE_TIMING_TESTS "Registrar benchmarks e metas de tempo do Stable Fluids no CTest" OFF)
function(add_fog_field_tests)
    enable_testing()
    add_executable(fog_field_tests fog_field_tests.cpp fog_field.cpp)
    add_executable(fog_color_tests fog_color_tests.cpp)
    add_executable(fog_stable_tests fog_stable_tests.cpp fog_stable.cpp)
    add_executable(follow_hysteresis_test follow_hysteresis_test.cpp)
    if(MSVC)
        target_compile_options(fog_field_tests PRIVATE /W4 /WX /utf-8)
        target_compile_options(fog_color_tests PRIVATE /W4 /WX /utf-8)
        target_compile_options(fog_stable_tests PRIVATE /W4 /WX /utf-8)
        target_compile_options(follow_hysteresis_test PRIVATE /W4 /WX /utf-8)
    else()
        target_compile_options(fog_field_tests PRIVATE -Wall -Wextra -Werror)
        target_compile_options(fog_color_tests PRIVATE -Wall -Wextra -Werror)
        target_compile_options(fog_stable_tests PRIVATE -Wall -Wextra -Werror)
        target_compile_options(follow_hysteresis_test PRIVATE -Wall -Wextra -Werror)
    endif()
    # Pasta com o wind_field.h do host: liga a comparação número a número das fórmulas do vento.
    set(WIND_FIELD_HOST_DIR "" CACHE PATH "soh/soh/Enhancements/Graphics/Weather do host")
    if(WIND_FIELD_HOST_DIR)
        target_include_directories(fog_field_tests PRIVATE "${WIND_FIELD_HOST_DIR}")
        target_compile_definitions(fog_field_tests PRIVATE FOG_WIND_HOST_PARITY)
    endif()
    add_test(NAME fog_field COMMAND fog_field_tests)
    add_test(NAME fog_color COMMAND fog_color_tests)
    add_test(NAME fog_stable COMMAND fog_stable_tests --checks-only)
    add_test(NAME follow_hysteresis COMMAND follow_hysteresis_test)
    if(FOG_STABLE_TIMING_TESTS)
        add_test(NAME fog_stable_timing COMMAND fog_stable_tests --timing-only)
        set_tests_properties(fog_stable_timing PROPERTIES LABELS "benchmark")
    endif()
endfunction()
if(FOG_FIELD_TESTS_ONLY)
    enable_testing()
    add_fog_field_tests()
    return()
endif()
set(OOT_NATIVE_SDK "" CACHE FILEPATH "CMake do SDK nativo OoT")
if(NOT EXISTS "${OOT_NATIVE_SDK}")
    message(FATAL_ERROR "Informe -DOOT_NATIVE_SDK=<SDK>/OotNativeSdk.cmake")
endif()
include("${OOT_NATIVE_SDK}")
if(TARGET LinkSpan::OotSdk)
    set(OOT_NATIVE_LAYOUT_ID "${LinkSpanOotSdk_LAYOUT_ID}")
endif()
if(NOT OOT_NATIVE_LAYOUT_ID OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Use um SDK nativo OoT de 64 bits com layout ID")
endif()
if(WIN32)
    set(FOGREMASTER_PLATFORM win64)
    set(FOGREMASTER_LIBRARY fogremaster.dll)
elseif(APPLE AND CMAKE_SYSTEM_NAME STREQUAL "Darwin")
    set(FOGREMASTER_PLATFORM macos)
    set(FOGREMASTER_LIBRARY fogremaster.dylib)
else()
    message(FATAL_ERROR "O provider tem configuração para Windows x64 e macOS de 64 bits")
endif()

option(FOGREMASTER_FLUID "Variante de prova com advecção semi-lagrangiana (GFX-UNIFY-001)" OFF)
add_library(fogremaster SHARED provider.cpp fog_field.cpp fog_stable.cpp)
if(FOGREMASTER_FLUID)
    target_compile_definitions(fogremaster PRIVATE FOGREMASTER_FLUID)
endif()
set_target_properties(fogremaster PROPERTIES PREFIX "")
if(TARGET LinkSpan::OotSdk)
    target_link_libraries(fogremaster PRIVATE LinkSpan::OotSdk)
else()
    target_include_directories(fogremaster SYSTEM PRIVATE ${OOT_NATIVE_INCLUDE_DIRS})
    target_compile_definitions(fogremaster PRIVATE ${OOT_NATIVE_COMPILE_DEFINITIONS})
endif()
if(MSVC)
    target_compile_options(fogremaster PRIVATE /utf-8 /Zc:preprocessor /W4)
endif()
set_target_properties(fogremaster PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "$<1:${CMAKE_BINARY_DIR}/mod/provider>"
    LIBRARY_OUTPUT_DIRECTORY "$<1:${CMAKE_BINARY_DIR}/mod/provider>")
configure_file(main.lua "${CMAKE_BINARY_DIR}/mod/main.lua" COPYONLY)
configure_file(manifest.toml.in "${CMAKE_BINARY_DIR}/mod/manifest.toml" @ONLY)
configure_file("../../LICENSE" "${CMAKE_BINARY_DIR}/mod/LICENSE" COPYONLY)

enable_testing()
add_fog_field_tests()
