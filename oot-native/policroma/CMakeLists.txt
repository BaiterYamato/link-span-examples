cmake_minimum_required(VERSION 3.26)
project(Policroma LANGUAGES CXX)

set(OOT_NATIVE_SDK "" CACHE FILEPATH "OotNativeSdk.cmake do host OoT")
set(POLICROMA_BRIDGE_DIR "" CACHE PATH "Diretório da ponte Policroma no host")
set(POLICROMA_GLSLANG_DIR "" CACHE PATH "Fonte do glslang 15.1.0")
set(POLICROMA_SPIRV_CROSS_DIR "" CACHE PATH "Fonte do SPIRV-Cross vulkan-sdk-1.4.304.0")
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Policroma exige Windows x64 e um host OoT DirectX 11 compatível")
endif()
if(NOT EXISTS "${OOT_NATIVE_SDK}" OR NOT EXISTS "${POLICROMA_BRIDGE_DIR}/PolicromaBridge.h")
    message(FATAL_ERROR "Informe OOT_NATIVE_SDK e POLICROMA_BRIDGE_DIR")
endif()
foreach(header PassSource.h PackHistory.h PackManifest.h)
    if(NOT EXISTS "${POLICROMA_BRIDGE_DIR}/${header}")
        message(FATAL_ERROR "POLICROMA_BRIDGE_DIR exige a ponte HDR/V2 completa: falta ${header}")
    endif()
endforeach()
include("${OOT_NATIVE_SDK}")
if(NOT TARGET LinkSpan::OotSdk OR NOT LinkSpanOotSdk_LAYOUT_ID)
    message(FATAL_ERROR "Use o SDK portátil correspondente ao host")
endif()
find_package(libzip CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)

if(NOT EXISTS "${POLICROMA_GLSLANG_DIR}/glslang/Public/ShaderLang.h" OR
   NOT EXISTS "${POLICROMA_SPIRV_CROSS_DIR}/spirv_hlsl.hpp")
    message(FATAL_ERROR "Informe POLICROMA_GLSLANG_DIR e POLICROMA_SPIRV_CROSS_DIR")
endif()

set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF)
set(BUILD_EXTERNAL OFF)
set(ENABLE_GLSLANG_BINARIES OFF)
set(ENABLE_HLSL OFF)
set(ENABLE_OPT OFF)
set(ENABLE_SPVREMAPPER OFF)
set(GLSLANG_TESTS OFF)
set(GLSLANG_ENABLE_INSTALL OFF)
add_subdirectory("${POLICROMA_GLSLANG_DIR}" glslang EXCLUDE_FROM_ALL)

set(SPIRV_CROSS_CLI OFF)
set(SPIRV_CROSS_ENABLE_TESTS OFF)
set(SPIRV_CROSS_ENABLE_C_API OFF)
set(SPIRV_CROSS_ENABLE_CPP OFF)
set(SPIRV_CROSS_ENABLE_MSL OFF)
set(SPIRV_CROSS_ENABLE_REFLECT OFF)
set(SPIRV_CROSS_ENABLE_UTIL OFF)
set(SPIRV_CROSS_SKIP_INSTALL ON)
set(CMAKE_POLICY_DEFAULT_CMP0077 NEW)
add_subdirectory("${POLICROMA_SPIRV_CROSS_DIR}" spirv-cross EXCLUDE_FROM_ALL)

add_library(policroma_catalog STATIC Catalog.cpp GlslFragment.cpp IrisPack.cpp)
target_compile_features(policroma_catalog PUBLIC cxx_std_20)
target_compile_definitions(policroma_catalog PUBLIC NOMINMAX)
target_include_directories(policroma_catalog PUBLIC "${POLICROMA_BRIDGE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(policroma_catalog PUBLIC libzip::zip nlohmann_json::nlohmann_json
    glslang::glslang glslang::glslang-default-resource-limits spirv-cross-hlsl)
if(MSVC)
    target_compile_options(policroma_catalog PRIVATE /utf-8 /Zc:preprocessor /W4)
endif()

add_library(policroma SHARED provider.cpp)
target_compile_features(policroma PRIVATE cxx_std_20)
target_link_libraries(policroma PRIVATE LinkSpan::OotSdk policroma_catalog)
if(MSVC)
    target_compile_options(policroma PRIVATE /utf-8 /Zc:preprocessor /W4)
    set_property(TARGET policroma PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
# Tradução Iris sem GPU: FXC + reflexão com as regras de interface do host.
enable_testing()
add_executable(policroma_iris_tests tests/iris_tests.cpp)
target_compile_features(policroma_iris_tests PRIVATE cxx_std_20)
target_link_libraries(policroma_iris_tests PRIVATE policroma_catalog d3d11 d3dcompiler dxguid)
if(MSVC)
    target_compile_options(policroma_iris_tests PRIVATE /utf-8 /Zc:preprocessor /W4)
    set_property(TARGET policroma_iris_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(iris_material_fixture "${CMAKE_CURRENT_SOURCE_DIR}/packs/IrisMaterial")
set(iris_material_zip "${CMAKE_CURRENT_BINARY_DIR}/IrisMaterial.zip")
set(iris_material_entries shaders/final.fsh shaders/final.vsh shaders/shaders.properties)
list(TRANSFORM iris_material_entries PREPEND "${iris_material_fixture}/" OUTPUT_VARIABLE iris_material_sources)
add_custom_command(OUTPUT "${iris_material_zip}"
    COMMAND "${CMAKE_COMMAND}" -E tar cf "${iris_material_zip}" --format=zip -- ${iris_material_entries}
    WORKING_DIRECTORY "${iris_material_fixture}"
    DEPENDS "${CMAKE_CURRENT_LIST_FILE}" ${iris_material_sources}
    VERBATIM)
add_custom_target(policroma_iris_fixture DEPENDS "${iris_material_zip}")
add_dependencies(policroma_iris_tests policroma_iris_fixture)
add_test(NAME policroma_iris COMMAND policroma_iris_tests "${CMAKE_CURRENT_SOURCE_DIR}/packs" "${iris_material_zip}")

add_executable(policroma_native_tests tests/native_tests.cpp)
target_compile_features(policroma_native_tests PRIVATE cxx_std_20)
target_link_libraries(policroma_native_tests PRIVATE policroma_catalog d3dcompiler dxguid)
if(MSVC)
    target_compile_options(policroma_native_tests PRIVATE /utf-8 /Zc:preprocessor /W4)
endif()
add_test(NAME policroma_native COMMAND policroma_native_tests)

set_target_properties(policroma PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "$<1:${CMAKE_BINARY_DIR}/mod/provider>")
configure_file(main.lua "${CMAKE_BINARY_DIR}/mod/main.lua" COPYONLY)
configure_file(manifest.toml.in "${CMAKE_BINARY_DIR}/mod/manifest.toml" @ONLY)
configure_file("../../LICENSE" "${CMAKE_BINARY_DIR}/mod/LICENSE" COPYONLY)
configure_file("${POLICROMA_GLSLANG_DIR}/LICENSE.txt" "${CMAKE_BINARY_DIR}/mod/LICENSE.glslang.txt" COPYONLY)
configure_file("${POLICROMA_SPIRV_CROSS_DIR}/LICENSE" "${CMAKE_BINARY_DIR}/mod/LICENSE.spirv-cross.txt" COPYONLY)
